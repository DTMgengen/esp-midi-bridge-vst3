cmake_minimum_required(VERSION 3.22)
project(ESPMidiBridge VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
if(MSVC)
  add_compile_options(/utf-8)
endif()
option(ESP_BUILD_PLUGIN "Build JUCE controller and tests" ON)
enable_testing()
add_library(esp_tracking STATIC Source/Dsp.cpp Source/StrictPitch.cpp Source/MidiEngine.cpp)
target_include_directories(esp_tracking PUBLIC Source)
if(MSVC)
  target_compile_options(esp_tracking PRIVATE /W4 /fp:precise)
endif()
foreach(pair IN ITEMS "pitch_modes|PitchModeTests" "strict_accuracy|StrictAccuracyTests" "pitch_fidelity|PitchFidelityTests")
  string(REPLACE "|" ";" pair "${pair}")
  list(GET pair 0 test_name)
  list(GET pair 1 source_name)
  add_executable(${test_name} Tests/${source_name}.cpp)
  target_link_libraries(${test_name} PRIVATE esp_tracking)
  add_test(NAME ${test_name} COMMAND ${test_name})
endforeach()
add_executable(midi_engine_tests Tests/MidiEngineTests.cpp)
target_link_libraries(midi_engine_tests PRIVATE esp_tracking)
add_test(NAME midi_engine COMMAND midi_engine_tests)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
  add_test(NAME tracking_provenance COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/verify_engine.py")
endif()
if(ESP_BUILD_PLUGIN)
  set(ESP_JUCE_SOURCE "" CACHE PATH "Optional existing JUCE source checkout")
  if(ESP_JUCE_SOURCE)
    add_subdirectory("${ESP_JUCE_SOURCE}" "${CMAKE_BINARY_DIR}/juce")
  else()
    include(FetchContent)
    FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
      GIT_TAG 29396c22c93392d6738e021b83196283d6e4d850 GIT_SHALLOW FALSE)
    FetchContent_MakeAvailable(JUCE)
  endif()
  set(ESP_FORMATS VST3 Standalone)
  if(APPLE)
    list(APPEND ESP_FORMATS AU)
  endif()
  juce_add_plugin(ESPMidiBridge
    COMPANY_NAME "Gengen Audio" BUNDLE_ID "audio.gengen.espmidibridge"
    PLUGIN_MANUFACTURER_CODE GgAu PLUGIN_CODE EmB1
    PRODUCT_NAME "ESP MIDI Bridge"
    IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT TRUE
    IS_MIDI_EFFECT FALSE EDITOR_WANTS_KEYBOARD_FOCUS FALSE
    COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Fx Tools
    FORMATS ${ESP_FORMATS})
  juce_generate_juce_header(ESPMidiBridge)
  target_sources(ESPMidiBridge PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp Source/DeviceOutput.cpp Source/MidiCapture.cpp)
  target_compile_definitions(ESPMidiBridge PUBLIC
    JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
  target_link_libraries(ESPMidiBridge PRIVATE esp_tracking juce::juce_audio_utils
    PUBLIC juce::juce_recommended_config_flags)
  juce_add_console_app(bridge_plugin_tests PRODUCT_NAME "Bridge Plugin Tests")
  target_sources(bridge_plugin_tests PRIVATE Tests/PluginTests.cpp)
  target_include_directories(bridge_plugin_tests PRIVATE Source "${CMAKE_BINARY_DIR}/ESPMidiBridge_artefacts/JuceLibraryCode")
  target_link_libraries(bridge_plugin_tests PRIVATE ESPMidiBridge juce::juce_audio_utils)
  add_test(NAME plugin COMMAND bridge_plugin_tests)
  juce_add_console_app(bridge_capture_tests PRODUCT_NAME "Bridge MIDI Capture Tests")
  target_sources(bridge_capture_tests PRIVATE Tests/CaptureTests.cpp)
  target_include_directories(bridge_capture_tests PRIVATE Source "${CMAKE_BINARY_DIR}/ESPMidiBridge_artefacts/JuceLibraryCode")
  target_link_libraries(bridge_capture_tests PRIVATE ESPMidiBridge juce::juce_audio_utils)
  add_test(NAME midi_capture COMMAND bridge_capture_tests)
  juce_add_console_app(bridge_vst3_tests PRODUCT_NAME "Bridge VST3 Integration Tests")
  target_sources(bridge_vst3_tests PRIVATE Tests/Vst3Tests.cpp)
  target_compile_definitions(bridge_vst3_tests PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
  target_link_libraries(bridge_vst3_tests PRIVATE juce::juce_audio_utils)
  add_test(NAME vst3_midi_output COMMAND bridge_vst3_tests "${CMAKE_BINARY_DIR}/ESPMidiBridge_artefacts/$<CONFIG>/VST3/ESP MIDI Bridge.vst3")
  if(WIN32)
    # JUCE 8.0.12 does not make its generated .rc depend on Info.txt.
    # Refresh version resources on incremental version changes as well.
    foreach(app IN ITEMS ESPMidiBridge bridge_plugin_tests bridge_capture_tests bridge_vst3_tests)
      get_target_property(info ${app} JUCE_INFO_FILE)
      get_target_property(generated ${app} JUCE_GENERATED_SOURCES_DIRECTORY)
      add_custom_command(OUTPUT "${generated}/${app}_resources.rc" APPEND DEPENDS "${info}")
    endforeach()
  endif()
endif()
